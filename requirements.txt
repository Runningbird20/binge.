# Project 3 requirements
#
# This repository is a Node.js/React/Express application, not a Python app.
# There are no Python package dependencies to install with pip.
# This file documents the runtime requirements, Node dependencies, and the
# commands needed to install and run the project.
#
# System requirements
# - Node.js 18+
# - npm 9+
#
# Local environment
# - The app expects REACT_APP_API_URL=http://localhost:5001 for local use.
# - The repository already includes a .env file with local development values.
#
# Install dependencies
#   npm install
#
# If the SQLite native module needs a rebuild on your machine
#   npm rebuild better-sqlite3
#
# Run the full app locally
#   npm start
#
# Run only the frontend
#   npm run start:client
#
# Run only the backend API
#   npm run server
#
# Build for production
#   npm run build
#
# Run tests
#   npm test
#
# Optional catalog/data import commands
#   npm run import:books
#   npm run import:books:bulk
#   npm run import:books:resume
#   npm run import:movies
#   npm run import:movies:bulk
#   npm run import:movies:resume
#   npm run import:tv
#   npm run import:tv:bulk
#   npm run import:tv:resume
#   npm run import:plex
#   npm run import:plex:bulk
#   npm run import:plex:resume
